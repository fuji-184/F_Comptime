use fcomptime::prelude::*;

init_comptime!();

fn non_comptime(i: i32) -> i32 {
    i + 10 * 10
}

#[comptime]
fn with_param_caller_no_need_to_type_source(i: i32) {
    source! {
        let res = i + 20 * 30;
        output!(raw, i, "second");
    }
}

#[comptime]
fn main() {

    source! {
        let become_comptime = non_comptime(10) * 10;
        output!(raw, become_comptime, "become_comptime");
    }

    call_scope! {
        let become_comptime = call!("become_comptime");
        println!("become_comptime: {}", become_comptime);

        let second = func!("with_param_caller_no_need_to_type_source", 20);
        println!("second: {}", second);
    }
    
}
